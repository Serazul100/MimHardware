returnString(
value==null
?""
:value
)

.replace(/&/g,"&amp;")

.replace(/</g,"&lt;")

.replace(/>/g,"&gt;")

.replace(/"/g,"&quot;")

.replace(/'/g,"&#039;");

}


functionlocalISODate(date){

returnnewDate(
date.getTime()-
date.getTimezoneOffset()*
60000
)
.toISOString()
.split("T")[0];
