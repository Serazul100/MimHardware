<script>
  function loadSettings() {
    google.script.run.withSuccessHandler(function(settings) {
      document.getElementById('companyName').value = settings.companyName || '';
      document.getElementById('companyAddress').value = settings.companyAddress || '';
      document.getElementById('companyPhone').value = settings.companyPhone || '';
      document.getElementById('taxRate').value = settings.taxRate || '';
      document.getElementById('currency').value = settings.currency || '';
      document.getElementById('invoicePrefix').value = settings.invoicePrefix || '';
      document.getElementById('appLanguage').value = settings.appLanguage || 'bn';
      document.getElementById('whatsappTemplate').value = settings.whatsappTemplate || '';
    }).getAllSettings();
  }

  function saveSettings() {
    const settings = {
      companyName: document.getElementById('companyName').value,
      companyAddress: document.getElementById('companyAddress').value,
      companyPhone: document.getElementById('companyPhone').value,
      taxRate: document.getElementById('taxRate').value,
      currency: document.getElementById('currency').value,
      invoicePrefix: document.getElementById('invoicePrefix').value,
      appLanguage: document.getElementById('appLanguage').value,
      whatsappTemplate: document.getElementById('whatsappTemplate').value
    };
    google.script.run.withSuccessHandler(function() {
      const msgDiv = document.getElementById('settingsMsg');
      msgDiv.innerHTML = '<span style="background: #064e3b; padding: 10px 20px; border-radius: 40px; display: inline-block;">✅ সেটিংস সংরক্ষিত হয়েছে!</span>';
      setTimeout(() => msgDiv.innerHTML = '', 3000);
    }).saveMultipleSettings(settings);
  }

  document.addEventListener('DOMContentLoaded', loadSettings);
</script>
