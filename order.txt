        function updateSelectionSummary() {
            const allCbs = Array.from(document.querySelectorAll('.invoice-checkbox'));
            const visibleCbs = allCbs.filter(cb => {
                const tr = cb.closest('tr');
                return tr && tr.style.display !== 'none';
            });
            const checkedVisible = visibleCbs.filter(cb => cb.checked);
            const checkedAll = allCbs.filter(cb => cb.checked);

            const bar = document.getElementById('selectionSummaryBar');

            if (checkedAll.length === 0) {
                bar.style.display = 'none';
            } else {
                bar.style.display = 'flex';
            }

            let total = 0, due = 0;
            checkedAll.forEach(cb => {
                total += parseFloat(cb.dataset.total) || 0;
                due   += parseFloat(cb.dataset.due)   || 0;
            });

            document.getElementById('selectedCount').innerText = checkedAll.length;
            document.getElementById('selectedTotal').innerText = total.toFixed(2);
            document.getElementById('selectedDue').innerText   = due.toFixed(2);

            const master = document.getElementById('selectAllInvoices');
            if (master) {
                if (visibleCbs.length === 0) {
                    master.checked = false;
                    master.indeterminate = false;
                } else if (checkedVisible.length === visibleCbs.length) {
                    master.checked = true;
                    master.indeterminate = false;
                } else if (checkedVisible.length === 0) {
                    master.checked = false;
                    master.indeterminate = false;
                } else {
                    master.checked = false;
                    master.indeterminate = true;
                }
            }
        }

        function printProductSummary() {
            const checked = Array.from(document.querySelectorAll('.invoice-checkbox:checked'));
            if (checked.length === 0) {
                alert('অনুগ্রহ করে কমপক্ষে একটি ইনভয়েস সিলেক্ট করুন!');
                return;
            }

            const productMap = {};

            checked.forEach(cb => {
                const invoiceNo = cb.dataset.invoice;
                const invoiceDate = String(cb.dataset.date || '').trim();
                const shopName = String(cb.dataset.shop || '').trim() || 'Unknown';
                const rows = orderDataCache[invoiceNo] || [];

                rows.forEach(r => {
                    const productName = String(r[5] || '').trim();
                    const color       = String(r[6] || '').trim();
                    const qty         = parseFloat(r[7]) || 0;
                    const total       = parseFloat(r[9]) || 0;
                    if (!productName) return;

                    const key = productName + '||' + color;

                    if (!productMap[key]) {
                        productMap[key] = {
                            productName: productName,
                            color: color,
                            totalQty: 0,
                            totalAmount: 0,
                            invoiceCount: 0,
                            _invoices: {},
                            _dates: {},
                            _shopQty: {}
                        };
                    }
                    const pm = productMap[key];
                    pm.totalQty    += qty;
                    pm.totalAmount += total;
                    if (!pm._invoices[invoiceNo]) {
                        pm._invoices[invoiceNo] = true;
                        pm.invoiceCount++;
                    }
                    if (invoiceDate) pm._dates[invoiceDate] = true;
                    if (!pm._shopQty[shopName]) pm._shopQty[shopName] = 0;
                    pm._shopQty[shopName] += qty;
                });
            });

            const productList = Object.values(productMap)
                .sort((a, b) => b.totalAmount - a.totalAmount);

            function prettyDate(iso) {
                const parts = String(iso).split('-');
                if (parts.length !== 3) return iso;
                const months = ['Jan','Feb','Mar','Apr','May','Jun','Jul','Aug','Sep','Oct','Nov','Dec'];
                return parts[2] + ' ' + (months[parseInt(parts[1], 10) - 1] || parts[1]);
            }
            function formatDateRange(dObj) {
                const ds = Object.keys(dObj).sort();
                if (ds.length === 0) return '-';
                if (ds.length === 1) return prettyDate(ds[0]);
                return prettyDate(ds[0]) + '–' + prettyDate(ds[ds.length - 1]);
            }

        function viewOrder(invoiceNo) {
            if (!invoiceNo) return;
            const rows = orderDataCache[invoiceNo];
            if (!rows || rows.length === 0) {
                alert('No data found for invoice: ' + invoiceNo);
                return;
            }

            const first = rows[0];
            let items = [];
            let grandTotal = 0;
            rows.forEach(r => {
                const qty = parseFloat(r[7]) || 0;
                const rate = parseFloat(r[8]) || 0;
                const total = parseFloat(r[9]) || 0;
                items.push({
                    productName: r[5] || '',
                    color: r[6] || '',
                    qty: qty,
                    rate: rate,
                    total: total
                });
                grandTotal += total;
            });
            const due = parseFloat(first[11]) || 0;
            const isPaid = due === 0;

            let shopName = first[3] || '';
            let address = '';
            if (shopName) {
                const cust = customersData.find(c => c.shopName === shopName);
                if (cust) address = cust.address || '';
            }

            const order = {
                invoice: first[0] || '',
                date: first[1] || '',
                customerName: first[2] || '',
                shopName: shopName,
                customerPhone: first[4] || '',
                address: address,
                items: items,
                grandTotal: grandTotal,
                due: due,
                isPaid: isPaid
            };
            currentViewOrder = order;
            renderViewModal(order);
        }

        function editOrder(invoiceNo) {
            if (!invoiceNo) return;
            const rows = orderDataCache[invoiceNo];
            if (!rows || rows.length === 0) {
                alert('No data found for invoice: ' + invoiceNo);
                return;
            }

            const first = rows[0];
            const items = [];
            let grandTotal = 0;
            rows.forEach(r => {
                const qty = parseFloat(r[7]) || 0;
                const rate = parseFloat(r[8]) || 0;
                const total = parseFloat(r[9]) || 0;
                items.push({
                    productName: r[5] || '',
                    color: r[6] || '',
                    details: r[6] || '',
                    qty: qty,
                    rate: rate,
                    total: total
                });
                grandTotal += total;
            });
            const due = parseFloat(first[11]) || 0;
            const isPaid = due === 0;

            document.getElementById('editModeFlag').value = 'true';
            document.getElementById('editOriginalInvoice').value = invoiceNo;
            document.getElementById('invoiceNo').value = invoiceNo;
            document.getElementById('orderDate').value = first[1] || '';
            document.getElementById('custShop').value = first[3] || '';
            document.getElementById('custName').value = first[2] || '';
            document.getElementById('custAddress').value = '';
            document.getElementById('custPhone').value = first[4] || '';
            document.getElementById('formPaymentToggle').checked = isPaid;

            const shop = first[3] || '';
            const cust = customersData.find(c => c.shopName === shop);
            document.getElementById('custId').value = cust ? (cust.id || '') : '';
            if (cust) {
                document.getElementById('custAddress').value = cust.address || '';
            }

            const tbody = document.querySelector('#orderTable tbody');
            tbody.innerHTML = '';
            if (items.length > 0) {
                items.forEach((item, idx) => {
                    const tr = document.createElement('tr');
                    tr.className = 'order-row';
                    tr.innerHTML = `
                        <td style="text-align: center;">${idx + 1}</td>
                        <td><input type="text" list="product_list" class="item-name" value="${esc(item.productName)}" oninput="fillProductData(this)"></td>

        function deleteRow(id) {
            if (!id) return;

                    })
                    .deleteOrder(id);
            }
        }

        //  SUBMIT / UPDATE ORDER
        // ============================================================
        function submitOrder() {
            const items = [];
            document.querySelectorAll('#orderTable tbody tr').forEach(tr => {
                let n = tr.querySelector('.item-name').value;
                if (n && n.trim()) {
                    items.push({
                        productName: n.trim(),
                        details: tr.querySelector('.item-color').value || '',
                        color: tr.querySelector('.item-color').value || '',
                        qty: tr.querySelector('.item-qty').value || 1,
                        rate: tr.querySelector('.item-rate').value || 0,
                        total: tr.querySelector('.item-total').innerText || '0'
                    });
                }
            });
            if (items.length === 0) { alert("পণ্য যোগ করুন!"); return; }

            const shopName = document.getElementById('custShop').value.trim();
            if (!shopName) { alert("দোকানের নাম লিখুন!"); return; }

            let rawPhone = document.getElementById('custPhone').value;
            if (rawPhone && !rawPhone.startsWith("'")) rawPhone = "'" + rawPhone;

            let isPaid = document.getElementById('formPaymentToggle').checked;
            let isEdit = document.getElementById('editModeFlag').value === 'true';
            let originalInvoice = document.getElementById('editOriginalInvoice').value;

            const orderData = {
                date: document.getElementById('orderDate').value,
                invoiceNo: document.getElementById('invoiceNo').value,
                customerId: document.getElementById('custId').value,
                customerName: document.getElementById('custName').value,
                shopName: shopName,
                customerPhone: rawPhone,
                grandTotal: document.getElementById('grandTotal').innerText,
                isPaid: isPaid,
                items: items,
                isEdit: isEdit,
                originalInvoice: originalInvoice
            };

            const handler = (res) => {
                if (res && res.success === false) {
                    alert('❌ সমস্যা: ' + (res.error || 'Unknown error'));
                    return;
                }
                alert(isEdit ? "✅ অর্ডার সফলভাবে আপডেট হয়েছে!" : "✅ অর্ডার সফলভাবে সেভ হয়েছে!");
                document.getElementById('editModeFlag').value = 'false';
                document.getElementById('editOriginalInvoice').value = '';
                document.getElementById('custId').value = '';
                switchOrderView('list');
            };

            if (isEdit) {
                google.script.run.withSuccessHandler(handler).updateOrderData(orderData);
            } else {
                google.script.run.withSuccessHandler(handler).saveOrderData(orderData);
            }
        }

        function filterTable() {
            const searchText = document.getElementById("searchField").value.toLowerCase().trim();
            const selectedDate = document.getElementById("searchDate").value;
            const rows = document.querySelectorAll("#orderTableBody tr");

            rows.forEach(row => {
                if (!row.dataset.invoice) return;
                const rowText = (row.innerText || '').toLowerCase();
                const rowDate = row.cells[2] ? row.cells[2].innerText.trim() : '';
                const matchText = !searchText || rowText.includes(searchText);
                const matchDate = !selectedDate || rowDate === selectedDate;
                row.style.display = (matchText && matchDate) ? '' : 'none';
            });

            updateSelectionSummary();
        }

        function detectColor(input) {
            let val = input.value.toLowerCase().trim();
            let colors = {
                "লাল": "red", "red": "red",
                "সাদা": "#a6a6a6", "white": "#a6a6a6",
                "নীল": "blue", "blue": "blue",
                "সবুজ": "green", "green": "green",
                "হলুদ": "#cca300", "yellow": "#cca300",
                "কালো": "black", "black": "black",
                "খয়েরি": "saddlebrown", "brown": "saddlebrown",
                "সোনালী": "goldenrod", "golden": "goldenrod",
                "কমলা": "darkorange", "orange": "darkorange",
                "ছাই": "gray", "grey": "gray",
                "বেগুনি": "purple", "purple": "purple",
                "গোলাপী": "hotpink", "pink": "hotpink"
            };
            let foundColor = colors[val] || "";
            if (foundColor) {
                input.style.borderLeft = "5px solid " + foundColor;
                input.style.color = foundColor;
                input.style.fontWeight = "bold";
            } else {
                input.style.borderLeft = "1px solid #ccc";
                input.style.color = "inherit";
                input.style.fontWeight = "normal";
            }
        }

        function calculateRow(e) {
            let tr = e.closest('tr');
            let qty = parseFloat(tr.querySelector('.item-qty').value) || 0;
            let rate = parseFloat(tr.querySelector('.item-rate').value) || 0;
            tr.querySelector('.item-total').innerText = (qty * rate).toFixed(2);
            calculateFinal();
        }

            document.getElementById('grandTotal').innerText = t.toFixed(2);
        }
