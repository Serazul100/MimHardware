/* SEARCH */
function applySearch() {
  var input = document.getElementById('salesSearch');
  var wrapper = document.getElementById('searchWrapper');
  if (input.value) wrapper.classList.add('has-value');
  else wrapper.classList.remove('has-value');

  var q = (input.value || '').trim().toLowerCase();
  var rows = document.querySelectorAll('#salesBody tr');
  var visibleCount = 0;
  var today = new Date().toISOString().split('T')[0];

  rows.forEach(function (tr) {
    var sale = cachedSalesData[tr.rowIndex - 1];
    if (!sale) { tr.classList.add('hidden-row'); return; }
    var match = true;

    if (currentQuickFilter === 'Paid') {
      var st = String(sale.status || '').toLowerCase();
      match = (st === 'paid' || st === 'confirmed');
    } else if (currentQuickFilter === 'Partial') {
      match = (String(sale.status || '').toLowerCase() === 'partial');
    } else if (currentQuickFilter === 'Due') {
      var st2 = String(sale.status || '').toLowerCase();
      match = (st2 === 'due' || st2 === 'pending');
    } else if (currentQuickFilter === 'today') {
      match = (String(sale.soDate || '').split('T')[0] === today);
    }

    if (match && q) {
      var haystack = [
        sale.soDate || '', sale.soId || '', sale.customerId || '',
        sale.customerName || '', sale.invoiceNum || '', sale.status || '',
        String(sale.totalAmount || ''), String(sale.totalReceived || ''),
        String(sale.soBalance || '')
      ].join(' ').toLowerCase();
      match = haystack.indexOf(q) !== -1;
    }

    tr.classList.toggle('hidden-row', !match);
    if (match) visibleCount++;
  });

  document.getElementById('searchCount').innerText = visibleCount;
  document.getElementById('noResults').style.display =
    (visibleCount === 0 && rows.length > 0) ? 'block' : 'none';
}

}

/* SORT */
function sortSalesByField(field, direction) {
  if (!cachedSalesData || cachedSalesData.length === 0) {
    alert("কোনো ডাটা নেই!");
    return;
  }
  var sorted = cachedSalesData.slice();
  sorted.sort(function (a, b) {
    var va = a[field], vb = b[field];

    if (field === 'totalAmount' || field === 'totalReceived' || field === 'soBalance') {
      va = parseFloat(va) || 0;
      vb = parseFloat(vb) || 0;
      return direction === 'asc' ? va - vb : vb - va;
    }
    if (field === 'soDate') {
      return direction === 'asc'
        ? String(va).localeCompare(String(vb))
        : String(vb).localeCompare(String(va));
    }
    if (field === 'soId') {
      var na = parseInt(va) || 0, nb = parseInt(vb) || 0;
      return direction === 'asc' ? na - nb : nb - na;
    }
    var sa = String(va || '').toLowerCase();
    var sb = String(vb || '').toLowerCase();
    return direction === 'asc' ? sa.localeCompare(sb) : sb.localeCompare(sa);
  });
  cachedSalesData = sorted;
  renderSales(cachedSalesData);
}


/* RENDER */
function renderSales(sales) {
  document.getElementById('loading').style.display = 'none';
  document.getElementById('salesTable').style.display = 'table';

  var tbody = document.getElementById('salesBody');
  tbody.innerHTML = '';

  sales.forEach(function (sale) {
    var tr = document.createElement('tr');

    var st = String(sale.status || 'Pending').trim();
    var stLower = st.toLowerCase();

    if (stLower === 'confirmed' || stLower === 'paid') {
      tr.className = 'row-paid';
    } else if (stLower === 'partial') {
      tr.className = 'row-partial';
    } else if (stLower === 'due' || stLower === 'pending') {
      tr.className = 'row-due';
    }

    var badgeClass = 'badge-pending';
    if (stLower === 'confirmed' || stLower === 'paid') badgeClass = 'badge-paid';
    else if (stLower === 'partial') badgeClass = 'badge-partial';
    else if (stLower === 'due' || stLower === 'pending') badgeClass = 'badge-due';

    var inv = String(sale.invoiceNum || '').replace(/'/g, "\\'");
    var isPaidStatus = (stLower === 'paid' || stLower === 'confirmed');
    var dueValue = parseFloat(sale.soBalance || 0);

    var actionHTML = '<td>';
    actionHTML += '<button class="btn-3d btn-edit" onclick="openEditPopup(\'' + inv + '\', \'' + (sale.soDate || '') + '\', \'' + (sale.status || 'Pending') + '\')"><i class="fas fa-pen"></i> Edit</button>';

    if (!isPaidStatus && dueValue > 0.01) {
      actionHTML += '<button class="btn-3d btn-payment" onclick="openReceivePay(\'' + inv + '\')"><i class="fas fa-money-bill-wave"></i> Payment</button>';
    } else {
      actionHTML += '<span class="btn-paid-badge"><i class="fas fa-check-circle"></i> Paid</span>';
    }

    if (!isPaidStatus) {
      actionHTML += '<button class="btn-3d btn-confirm" onclick="confirmOrder(\'' + inv + '\')"><i class="fas fa-check"></i> Confirm</button>';
    }

    actionHTML += '<button class="btn-3d btn-remove" onclick="removeOrder(\'' + inv + '\')"><i class="fas fa-trash"></i></button>';
    actionHTML += '</td>';

    tr.innerHTML =
      '<td>' + (sale.soDate || '') + '</td>' +
      '<td style="font-weight:800; color:var(--accent);">' + (sale.soId || '') + '</td>' +
      '<td>' + (sale.customerId || '') + '</td>' +
      '<td style="font-weight:600;">' + (sale.customerName || '') + '</td>' +
      '<td>' + (sale.invoiceNum || '') + '</td>' +
      '<td style="font-weight:700;">৳ ' + parseFloat(sale.totalAmount || 0).toFixed(2) + '</td>' +
      '<td style="font-weight:700; color:#34d399;">৳ ' + parseFloat(sale.totalReceived || 0).toFixed(2) + '</td>' +
      '<td style="font-weight:700; color:' + (dueValue > 0 ? '#f87171' : '#94a3b8') + ';">৳ ' + dueValue.toFixed(2) + '</td>' +
      '<td><span class="badge-status ' + badgeClass + '">' + (sale.status || 'Pending') + '</span></td>' +
      actionHTML;

    tbody.appendChild(tr);
  });

  document.getElementById('searchCount').innerText = sales.length;
  applySearch();
}

    var rate = parseFloat(row[8]) || 0;
    var total = parseFloat(row[9]) || (qty * rate);

    if (pName) {
      var key = String(pName).trim().toLowerCase() + "_" + (pColor ? String(pColor).trim().toLowerCase() : "");
      if (!summaryMap[key]) {
        summaryMap[key] = { name: pName, color: pColor || "N/A", qty: 0, total: 0 };
      }
      summaryMap[key].qty += qty;
      summaryMap[key].total += total;
    }
  });

  var index = 1, finalGrandTotal = 0;
  for (var k in summaryMap) {
    var data = summaryMap[k];
    var tr = document.createElement('tr');
    tr.innerHTML =
      '<td style="text-align:center; font-weight:bold;">' + (index++) + '</td>' +
      '<td style="font-weight:600; color:var(--text);">' + data.name + '</td>' +
      '<td style="text-align:center; color:var(--muted);">' + data.color + '</td>' +
      '<td style="text-align:center; font-weight:bold; color:#34d399;">' + data.qty + '</td>' +
      '<td style="text-align:right; font-weight:bold; color:var(--accent);">' + data.total.toFixed(2) + ' ৳</td>';
    tbody.appendChild(tr);
    finalGrandTotal += data.total;
  }
  document.getElementById('grandTotal').innerText = finalGrandTotal.toFixed(2);
  window.currentEditingInvoice = "SUMMARY_" + document.getElementById('summaryDateFilter').value;
}


/* SAVE ORDER */
function submitOrder() {
  var invoiceNo = window.currentEditingInvoice;
  if (!invoiceNo) return alert("ইনভয়েস নম্বর পাওয়া যায়নি!");

  var customerInfo = {
    date: document.getElementById('popupDate').innerText,
    shop: document.getElementById('custShop').value.trim(),
    name: document.getElementById('custName').value.trim(),
    address: document.getElementById('custAddress').value.trim(),
    phone: document.getElementById('custPhone').value.trim()
  };
  if (!customerInfo.shop) return alert("দোকানের নাম দরকার!");

  var itemsList = [];
  document.querySelectorAll('#orderItemsBody .order-row').forEach(function (row) {
    var name = row.querySelector('.item-name').value;
    if (name && name.trim()) {
      itemsList.push({
        name: name.trim(),
        color: row.querySelector('.item-color').value,
        qty: parseFloat(row.querySelector('.item-qty').value) || 0,
        rate: parseFloat(row.querySelector('.item-rate').value) || 0
      });
    }
  });

  if (itemsList.length === 0) return alert("অন্তত একটি প্রোডাক্ট দিন!");

  if (typeof google !== 'undefined' && google.script) {
    var btn = document.getElementById('saveOrderBtn');
    var oldText = btn.innerHTML;
    btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Saving...';
    btn.disabled = true;

    google.script.run
      .withSuccessHandler(function (res) {
        btn.innerHTML = oldText;
        btn.disabled = false;
        if (res && String(res).indexOf('Error') === 0) {
          alert("❌ " + res);
          return;
        }
        alert("✅ ইনভয়েস সফলভাবে সংরক্ষণ হয়েছে!");
        closePopup();
        loadSales();
      })
      .withFailureHandler(function (err) {
        btn.innerHTML = oldText;
        btn.disabled = false;
        alert("❌ ব্যর্থ: " + err);
      })
      .updateOrderData(invoiceNo, customerInfo, itemsList);
  } else {
    alert("Local Save Success");
    closePopup();
  }
}

function reloadStockReport() {
  var fromDate = document.getElementById('srFromDate').value;
  var toDate = document.getElementById('srToDate').value;
  var filterType = document.getElementById('srFilterType').value;
  var sortBy = document.getElementById('srSortBy').value;
  var sortDir = document.getElementById('srSortDir').value;

  if (!fromDate || !toDate) {
    alert("From Date এবং To Date দুটোই দিন!");
    return;
  }

  if (fromDate > toDate) {
    var tmp = fromDate;
    fromDate = toDate;
    toDate = tmp;
    document.getElementById('srFromDate').value = fromDate;
    document.getElementById('srToDate').value = toDate;
  }

  var label = (fromDate === toDate)
    ? 'তারিখ: ' + fromDate
    : 'রেঞ্জ: ' + fromDate + ' → ' + toDate;
  document.getElementById('srDateLabel').innerText = label;

  var tbody = document.getElementById('srTableBody');

        if (!res || !res.success) {

function openReceivePay(invoiceNo) {
  currentReceiveInvoice = invoiceNo;

  var sale = (cachedSalesData || []).find(function (s) {
    return String(s.invoiceNum) === String(invoiceNo);
  });

  if (!sale) { alert("Invoice পাওয়া যায়নি!"); return; }

  document.getElementById('rpInvoice').innerText = invoiceNo;
  document.getElementById('rpCustomer').innerText = sale.customerName || '-';
  document.getElementById('rpTotal').innerText = parseFloat(sale.totalAmount || 0).toFixed(2);
  document.getElementById('rpPaid').innerText = parseFloat(sale.totalReceived || 0).toFixed(2);
  document.getElementById('rpDue').innerText = parseFloat(sale.soBalance || 0).toFixed(2);
  document.getElementById('rpStatus').innerText = sale.status || 'Due';

  document.getElementById('rpAmount').value = '';
  document.getElementById('rpMethod').value = 'Cash';
  document.getElementById('rpNote').value = '';

  document.getElementById('rpHistoryBody').innerHTML =
    '<tr><td colspan="5" style="color:var(--muted);">লোড হচ্ছে...</td></tr>';

  document.getElementById('receivePayModal').classList.add('show');

  checkPaymentLock(sale);

  var amountInput = document.getElementById('rpAmount');
  amountInput.oninput = function() {
    validatePaymentAmount(parseFloat(this.value) || 0, sale);
  };

  if (typeof google !== 'undefined' && google.script) {
    google.script.run.withSuccessHandler(function (rows) {
      renderPayHistory(rows || []);
    }).getPaymentsByInvoice(invoiceNo);
  } else {
    renderPayHistory([]);
  }
}

function checkPaymentLock(sale) {
  var due = parseFloat(sale.soBalance || 0);
  var total = parseFloat(sale.totalAmount || 0);
  var paid = parseFloat(sale.totalReceived || 0);
  var saveBtn = document.querySelector('#receivePayModal .btn-pay-save');
  var amountInput = document.getElementById('rpAmount');
  var warningBox = document.getElementById('rpWarningBox');

  if (!warningBox) {
    warningBox = document.createElement('div');
    warningBox.id = 'rpWarningBox';
    warningBox.style.cssText =
      'display:none; padding:12px 16px; border-radius:10px; margin-bottom:12px;' +
      'font-size:13px; font-weight:600; line-height:1.5;';
    var amountRow = amountInput.closest('.pay-input-row');
    if (amountRow) amountRow.parentNode.insertBefore(warningBox, amountRow);
  }

  if (due <= 0.01) {
    if (saveBtn) {
      saveBtn.disabled = true;
      saveBtn.style.opacity = '0.4';
      saveBtn.style.cursor = 'not-allowed';
      saveBtn.innerHTML = '<i class="fas fa-lock"></i> Payment Complete';
    }
    if (amountInput) {
      amountInput.disabled = true;
      amountInput.style.opacity = '0.5';
      amountInput.style.cursor = 'not-allowed';
      amountInput.placeholder = 'সম্পূর্ণ পরিশোধিত';
    }
    warningBox.style.display = 'block';
    warningBox.style.background = 'rgba(16,185,129,0.12)';
    warningBox.style.color = '#34d399';
    warningBox.style.border = '1px solid rgba(16,185,129,0.35)';
    warningBox.innerHTML =
      '<i class="fas fa-check-circle"></i> ' +
      '<strong>এই Invoice সম্পূর্ণ পরিশোধিত।</strong><br>' +
      'মোট ৳' + total.toFixed(2) + ' এর মধ্যে ৳' + paid.toFixed(2) +
      ' গ্রহণ করা হয়েছে। আর কোনো পেমেন্ট নেওয়া যাবে না।';
  } else {
    if (saveBtn) {
      saveBtn.disabled = false;
      saveBtn.style.opacity = '1';
      saveBtn.style.cursor = 'pointer';
      saveBtn.innerHTML = '<i class="fas fa-floppy-disk"></i> Save Payment';
    }
    if (amountInput) {
      amountInput.disabled = false;
      amountInput.style.opacity = '1';
      amountInput.style.cursor = 'auto';
      amountInput.placeholder = '0.00';
    }
    warningBox.style.display = 'none';
  }
}

function validatePaymentAmount(amount, sale) {
  var due = parseFloat(sale.soBalance || 0);
  var warningBox = document.getElementById('rpWarningBox');
  var saveBtn = document.querySelector('#receivePayModal .btn-pay-save');

  if (due <= 0.01) return;

  if (amount <= 0) {
    if (warningBox) warningBox.style.display = 'none';
    if (saveBtn) {
      saveBtn.disabled = false;
      saveBtn.style.opacity = '1';
      saveBtn.style.cursor = 'pointer';
      saveBtn.innerHTML = '<i class="fas fa-floppy-disk"></i> Save Payment';
    }
    return;
  }

  if (amount > due + 0.01) {
    if (warningBox) {
      warningBox.style.display = 'block';
      warningBox.style.background = 'rgba(239,68,68,0.12)';
      warningBox.style.color = '#f87171';
      warningBox.style.border = '1px solid rgba(239,68,68,0.35)';
      warningBox.innerHTML =
        '<i class="fas fa-exclamation-triangle"></i> ' +
        '<strong>সতর্কতা:</strong> বাকি আছে ৳' + due.toFixed(2) +
        ', কিন্তু আপনি ৳' + amount.toFixed(2) + ' দিচ্ছেন।<br>' +
        'সর্বোচ্চ ৳' + due.toFixed(2) + ' নেওয়া যাবে।';
    }
    if (saveBtn) {
      saveBtn.disabled = true;
      saveBtn.style.opacity = '0.4';
      saveBtn.style.cursor = 'not-allowed';
      saveBtn.innerHTML = '<i class="fas fa-lock"></i> Amount বেশি';
    }
  } else {
    if (warningBox) {
      warningBox.style.display = 'block';
      warningBox.style.background = 'rgba(56,189,248,0.12)';
      warningBox.style.color = 'var(--accent)';
      warningBox.style.border = '1px solid var(--accent)';
      warningBox.innerHTML =
        '<i class="fas fa-info-circle"></i> ' +
        '৳' + amount.toFixed(2) + ' নিলে বাকি থাকবে ৳' +
        (due - amount).toFixed(2) + '।';
    }
    if (saveBtn) {
      saveBtn.disabled = false;
      saveBtn.style.opacity = '1';
      saveBtn.style.cursor = 'pointer';
      saveBtn.innerHTML = '<i class="fas fa-floppy-disk"></i> Save Payment';
    }
  }
}

function renderPayHistory(rows) {
  var tbody = document.getElementById('rpHistoryBody');
  tbody.innerHTML = '';
  if (!rows || rows.length === 0) {
    tbody.innerHTML = '<tr><td colspan="5" style="color:var(--muted);">এখনো কোনো পেমেন্ট নেই</td></tr>';
    return;
  }
  rows.forEach(function (r, i) {
    var tr = document.createElement('tr');
    tr.innerHTML =
      '<td>' + (i + 1) + '</td>' +
      '<td>' + (r.date || '') + '</td>' +
      '<td style="text-align:right; font-weight:bold;">৳ ' + parseFloat(r.amount || 0).toFixed(2) + '</td>' +
      '<td>' + (r.method || 'Cash') + '</td>' +
      '<td style="font-size:11px;">' + (r.note || '') + '</td>';
    tbody.appendChild(tr);
  });
}

function saveReceivePayment() {
  var invoiceNo = currentReceiveInvoice;
  if (!invoiceNo) return;

  var amount = parseFloat(document.getElementById('rpAmount').value) || 0;
  if (amount <= 0) { alert("সঠিক টাকার পরিমাণ দিন!"); return; }

  var method = document.getElementById('rpMethod').value;
  var note   = document.getElementById('rpNote').value.trim();

  var saleCheck = (cachedSalesData || []).find(function (s) {
    return String(s.invoiceNum) === String(invoiceNo);
  });
  if (saleCheck) {
    var dueCheck = parseFloat(saleCheck.soBalance || 0);
    if (dueCheck <= 0.01) {
      alert("❌ এই Invoice সম্পূর্ণ পরিশোধিত। আর পেমেন্ট নেওয়া যাবে না।");
      return;
    }
    if (amount > dueCheck + 0.01) {
      alert("❌ বাকি আছে মাত্র ৳" + dueCheck.toFixed(2) + "। এর বেশি নেওয়া যাবে না।");
      return;
    }
  }

  var btn = event.target;
  var oldText = btn.innerHTML;
  btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Saving...';
  btn.disabled = true;

  google.script.run
    .withSuccessHandler(function (res) {
      btn.innerHTML = oldText;
      btn.disabled = false;
      if (!res || !res.success) {
        alert("❌ " + (res && res.error ? res.error : "ব্যর্থ"));
        return;
      }
      alert("✅ পেমেন্ট সংরক্ষণ হয়েছে!");
      closeReceivePay();
      loadSales();
    })
    .withFailureHandler(function (err) {
      btn.innerHTML = oldText;
      btn.disabled = false;
      alert("❌ " + err);
    })
    .addPartialPayment(invoiceNo, amount, method, note);
}

  document.getElementById('receivePayModal').classList.remove('show');
  currentReceiveInvoice = null;

  var amountInput = document.getElementById('rpAmount');
  if (amountInput) {

    amountInput.style.cursor = 'auto';

    amountInput.placeholder = '0.00';

  }

  var warningBox = document.getElementById('rpWarningBox');

  var saveBtn = document.querySelector('#receivePayModal .btn-pay-save');
  if (saveBtn) {
    saveBtn.disabled = false;
    saveBtn.style.opacity = '1';
    saveBtn.style.cursor = 'pointer';
    saveBtn.innerHTML = '<i class="fas fa-floppy-disk"></i> Save Payment';
  }
}

function calculatePITotals(e) {

  var rate = parseFloat(tr.querySelector('.pi-rate').value) || 0;
  tr.querySelector('.pi-total').innerText = (qty * rate).toFixed(2);
  updatePIGrandTotal();
}

function updatePIGrandTotal() {

   ========================================================= */
function openCustomerStatementModal() {
  var select = document.getElementById('statementCustomerSelect');
  select.innerHTML = '<option value="">গ্রাহকের তালিকা লোড হচ্ছে...</option>';
  document.getElementById('customerStatementModal').classList.add('show');
  if (!(typeof google !== 'undefined' && google.script)) {
    select.innerHTML = '<option value="">Google Apps Script সংযোগ পাওয়া যায়নি</option>';
    return;
  }
  google.script.run.withSuccessHandler(function(list) {
    var customers = list || [];
    select.innerHTML = '<option value="">-- গ্রাহক নির্বাচন করুন --</option>';
    customers.forEach(function(c) {
      if (!c) return;
      var id = String(c.id || '').trim();
      if (!id) return;
      var shop = c.shop || c.shopName || '';
      var label = [shop || c.name || 'Customer', c.name, c.phone, id].filter(Boolean).join(' — ');
      var option = document.createElement('option');
      option.value = id;
      option.textContent = label;
      select.appendChild(option);
    });
    if (select.options.length === 1) {
      select.innerHTML = '<option value="">Customers sheet-এ Customer ID/Customers ID সহ গ্রাহক পাওয়া যায়নি</option>';
    }
  }).withFailureHandler(function(err) {
    select.innerHTML = '<option value="">গ্রাহক লোড হয়নি: ' + String(err) + '</option>';
  }).getCustomersList();
}
function closeCustomerStatementModal() {
  document.getElementById('customerStatementModal').classList.remove('show');
}
function printCustomerStatement() {
  var customerId = document.getElementById('statementCustomerSelect').value;
  if (!customerId) return alert('আগে একজন গ্রাহক নির্বাচন করুন।');
  if (!(typeof google !== 'undefined' && google.script)) return alert('এই ফিচারটি Google Apps Script Web App-এ চালু করতে হবে।');
  var btn = document.querySelector('#customerStatementModal .btn-3d-print');
  var oldText = btn.innerHTML;
  btn.disabled = true;
  btn.innerHTML = '<i class="fas fa-spinner fa-spin"></i> Loading...';
  google.script.run.withSuccessHandler(function(res) {
    btn.disabled = false; btn.innerHTML = oldText;
    if (!res || !res.success) return alert('হিসাব লোড করা যায়নি: ' + ((res && res.error) || 'Unknown error'));
    closeCustomerStatementModal();
    buildCustomerStatementPrint(res);
  }).withFailureHandler(function(err) {
    btn.disabled = false; btn.innerHTML = oldText; alert('Error: ' + err);
  }).getCustomerStatement(customerId);
}

function escapeHtml(str) {
  return String(str == null ? '' : str)
    .replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;')
    .replace(/"/g, '&quot;').replace(/'/g, '&#39;');
}
