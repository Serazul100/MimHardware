<script>
'use strict';

var allSuppliers = [];
var currentEdit = null;
var isSaving = false;
var isDeleting = false;
var loadRequestId = 0;

/* ---------------- HELPERS ---------------- */

function safeNum(v) {
  if (v === null || v === undefined || v === '') return 0;
  var n = Number(v);
  return isFinite(n) ? n : 0;
}

function money(v) {
  return '৳ ' + safeNum(v).toLocaleString('en-BD', {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2
  });
}

function escapeHTML(v) {
  return String(v == null ? '' : v).replace(/[&<>"']/g, function (c) {
    return {
      '&': '&amp;', '<': '&lt;', '>': '&gt;',
      '"': '&quot;', "'": '&#39;'
    }[c];
  });
}

function showError(msg) {
  alert(msg || 'An unexpected error occurred.');
}

function setTableMessage(msg, color) {
  document.getElementById('tb').innerHTML =
    '<tr><td colspan="8" class="status-message" style="color:' +
    escapeHTML(color || '#94a3b8') + '">' +
    escapeHTML(msg) + '</td></tr>';
}

/* ---------------- LOAD ---------------- */

function loadSuppliers() {
  var reqId = ++loadRequestId;

  setTableMessage('Loading suppliers list...', '#38bdf8');

  google.script.run
    .withSuccessHandler(function (response) {
      if (reqId !== loadRequestId) return;

      console.log('=== getSupplierList RESPONSE ===');
      console.log('Is Array:', Array.isArray(response));
      console.log('Length:', Array.isArray(response) ? response.length : 'N/A');
      if (Array.isArray(response) && response.length > 0) {
        console.log('First row:', response[0]);
        console.log('First row purchases:', response[0].purchases);
        console.log('First row payments:', response[0].payments);
      }

      allSuppliers = Array.isArray(response) ? response : [];
      applySearchLocally();
    })
    .withFailureHandler(function (err) {
      if (reqId !== loadRequestId) return;
      setTableMessage(
        'Error: ' + (err && err.message ? err.message : 'Unknown'),
        '#f87171'
      );
    })
    .getSupplierList();   // ✅ Changed from supGetSuppliers()
}

/* ---------------- RENDER ---------------- */

function renderSuppliers(list) {
  var tbody = document.getElementById('tb');

  if (!Array.isArray(list) || list.length === 0) {
    setTableMessage('No suppliers found.');
    return;
  }

  var html = '';

  list.forEach(function (s) {
    var id      = String(s.id || s.supplierId || s.supplierID || '');
    var name    = String(s.name || s.supplierName || s.suppliername ||
                          s.supplier_name || s.supplier || '');
    var contact = String(s.contact || s.supplierContact || s.supplierPhone || '');
    var address = String(s.address || s.supplierAddress || s.supplier_address || '');

    var purchases = safeNum(s.purchases || s.totalPurchases || 0);
    var payments  = safeNum(s.payments  || s.totalPayments  || 0);
    var balance   = safeNum(s.balance   || s.balancePayable ||
                            (purchases - payments));

    var balanceColor = balance > 0 ? '#f87171'
                     : balance < 0 ? '#fbbf24'
                     : '#4ade80';

    html += '<tr>' +
      '<td><strong style="color:#fbbf24">' + escapeHTML(id) + '</strong></td>' +
      '<td>' + escapeHTML(name) + '</td>' +
      '<td>' + escapeHTML(contact) + '</td>' +
      '<td>' + escapeHTML(address) + '</td>' +
      '<td style="color:#38bdf8;font-weight:700;white-space:nowrap">' + money(purchases) + '</td>' +
      '<td style="color:#4ade80;font-weight:700;white-space:nowrap">' + money(payments) + '</td>' +
      '<td style="color:' + balanceColor + ';font-weight:800;white-space:nowrap">' + money(balance) + '</td>' +
      '<td style="white-space:nowrap">' +
        '<button type="button" class="btn edit" data-action="edit" data-id="' + escapeHTML(id) + '">Edit</button>' +
        '<button type="button" class="btn del" data-action="delete" data-id="' + escapeHTML(id) + '">Del</button>' +
      '</td>' +
    '</tr>';
  });

  tbody.innerHTML = html;
}

/* ---------------- SEARCH ---------------- */

function applySearchLocally() {
  var q = document.getElementById('supplierSearch').value.trim().toLowerCase();

  if (!q) {
    renderSuppliers(allSuppliers);
    return;
  }

  var filtered = allSuppliers.filter(function (s) {
    var id = s.id || s.supplierId || s.supplierID || '';
    var name = s.name || s.supplierName || s.suppliername || s.supplier_name || s.supplier || '';
    var contact = s.contact || s.supplierContact || '';
    var address = s.address || s.supplierAddress || '';
    return [id, name, contact, address].some(function (v) {
      return String(v == null ? '' : v).toLowerCase().indexOf(q) !== -1;
    });
  });

  renderSuppliers(filtered);
}

function searchSuppliers() {
  applySearchLocally();
}

/* ---------------- MODAL ---------------- */

function openNewSupplierModal() {
  currentEdit = null;
  document.getElementById('supplierForm').reset();
  document.getElementById('supplierId').readOnly = false;
  document.getElementById('modalTitle').textContent = 'Add New Supplier';
  document.getElementById('supplierModal').style.display = 'flex';
  document.getElementById('supplierName').focus();
}

function closeModal() {
  if (isSaving) return;
  document.getElementById('supplierModal').style.display = 'none';
  currentEdit = null;
}

function editSupplier(id) {
  var s = allSuppliers.filter(function (x) {
    var xid = x.id || x.supplierId || x.supplierID || '';
    return String(xid) === String(id);
  })[0];

  if (!s) {
    showError('Supplier not found. Please refresh and try again.');
    return;
  }

  currentEdit = String(id);

  var sid     = s.id || s.supplierId || s.supplierID || '';
  var sname   = s.name || s.supplierName || s.suppliername || s.supplier_name || s.supplier || '';
  var scont   = s.contact || s.supplierContact || '';
  var saddr   = s.address || s.supplierAddress || '';

  document.getElementById('supplierId').value = String(sid);
  document.getElementById('supplierId').readOnly = true;
  document.getElementById('supplierName').value = String(sname);
  document.getElementById('contact').value = String(scont);
  document.getElementById('address').value = String(saddr);
  document.getElementById('paymentDelta').value = '';
  document.getElementById('modalTitle').textContent = 'Modify Supplier Settings';
  document.getElementById('supplierModal').style.display = 'flex';
}

/* ---------------- SAVE ---------------- */

function saveSupplier(ev) {
  if (ev) ev.preventDefault();
  if (isSaving) return;

  var payload = {
    supplierId:      document.getElementById('supplierId').value.trim(),
    supplierName:    document.getElementById('supplierName').value.trim(),
    supplierContact: document.getElementById('contact').value.trim(),
    supplierAddress: document.getElementById('address').value.trim(),
    paymentDelta:    document.getElementById('paymentDelta').value.trim()
  };

  if (!payload.supplierName) {
    showError('Supplier Name is required.');
    document.getElementById('supplierName').focus();
    return;
  }

  isSaving = true;
  var btn = document.getElementById('saveBtn');
  btn.disabled = true;
  btn.textContent = 'Saving...';

  google.script.run
    .withSuccessHandler(function (res) {
      isSaving = false;
      btn.disabled = false;
      btn.textContent = 'Save Data';

      if (res && res.success) {
        document.getElementById('supplierModal').style.display = 'none';
        document.getElementById('supplierSearch').value = '';
        currentEdit = null;
        loadSuppliers();
      } else {
        showError('Save failed: ' +
          (res && res.error ? res.error : 'Unknown server error'));
      }
    })
    .withFailureHandler(function (err) {
      isSaving = false;
      btn.disabled = false;
      btn.textContent = 'Save Data';
      showError('Network error: ' +
        (err && err.message ? err.message : 'Unknown'));
    })
    .saveSupplierData(payload);
}

/* ---------------- DELETE ---------------- */

function deleteSupplier(id, button) {
  if (isDeleting) return;

  var s = allSuppliers.filter(function (x) {
    var xid = x.id || x.supplierId || x.supplierID || '';
    return String(xid) === String(id);
  })[0];

  if (!s) {
    showError('Supplier not found. Please refresh.');
    return;
  }

  var sname = s.name || s.supplierName || s.suppliername || s.supplier_name || s.supplier || '';
  var msg = 'Are you sure you want to delete this supplier?\n\n' +
            'ID: ' + String(id) + '\n' +
            'Name: ' + String(sname);

  if (!confirm(msg)) return;

  isDeleting = true;
  if (button) {
    button.disabled = true;
    button.textContent = 'Deleting...';
  }

  google.script.run
    .withSuccessHandler(function (res) {
      isDeleting = false;

      if (res && res.success) {
        allSuppliers = allSuppliers.filter(function (x) {
          var xid = x.id || x.supplierId || x.supplierID || '';
          return String(xid) !== String(id);
        });
        applySearchLocally();
        loadSuppliers();
      } else {
        if (button) {
          button.disabled = false;
          button.textContent = 'Del';
        }
        showError('Delete failed: ' +
          (res && res.error ? res.error : 'Unknown'));
      }
    })
    .withFailureHandler(function (err) {
      isDeleting = false;
      if (button) {
        button.disabled = false;
        button.textContent = 'Del';
      }
      showError('Network error: ' +
        (err && err.message ? err.message : 'Unknown'));
    })
    .supDeleteSupplier(id);
}

/* ---------------- EVENT LISTENERS ---------------- */

document.getElementById('addSupplierBtn')
  .addEventListener('click', openNewSupplierModal);

document.getElementById('closeModalBtn')
  .addEventListener('click', closeModal);

document.getElementById('supplierForm')
  .addEventListener('submit', saveSupplier);

document.getElementById('supplierSearch')
  .addEventListener('input', searchSuppliers);

document.getElementById('tb')
  .addEventListener('click', function (ev) {
    var btn = ev.target.closest('button[data-action]');
    if (!btn) return;
    var id = btn.dataset.id;
    var action = btn.dataset.action;
    if (action === 'edit') editSupplier(id);
    else if (action === 'delete') deleteSupplier(id, btn);
  });

document.getElementById('supplierModal')
  .addEventListener('click', function (ev) {
    if (ev.target === this) closeModal();
  });

document.addEventListener('keydown', function (ev) {
  if (ev.key === 'Escape') closeModal();
});

document.addEventListener('DOMContentLoaded', loadSuppliers);
</script>
